import { Card, CardGrid, LinkCard } from '@astrojs/starlight/components';

`aspire deploy` is hijacked so your distributed app ships to Cloudflare — the same way the AWS integration targets AWS instead of Azure.

<CardGrid>
  <Card title="R2 object storage" icon="cloud-download">
    Provision buckets via the Cloudflare API, with a local **MinIO** emulator for a
    credential-free dev loop. Plus a client that registers a R2-tuned `IAmazonS3`.
  </Card>
  <Card title="Workers" icon="rocket">
    Run hand-authored Workers locally with `wrangler dev` (no account needed) and
    deploy them with `aspire deploy`.
  </Card>
  <Card title="Containers" icon="seti:docker">
    Build a container image from your .NET project and run it on Cloudflare
    Containers — config and connection strings injected automatically.
  </Card>
  <Card title="Pages" icon="laptop">
    Attach to your Aspire JavaScript app and deploy the build output to Cloudflare
    Pages, with custom-domain support.
  </Card>
</CardGrid>
